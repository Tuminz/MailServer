*SELECTIONSORT
void selectionsort(int a[], int n)
{
for(int i=0; i<n;i++)
{
int min_pos=i;
for(int j=i+1;j<n;j++)
{
if(a[j]>a[min_pos])
{ min_pos=j; }
}
swap(a[i], a[min_pos]);
}
}
*BUBBLESORT:
void bubblesortI(int a[], int n)
{
for(int i=0;i<n-1;i++)
{
for(int j=0; j<n-i-1; j++)
{
if(a[j]>a[j+1])
{
swap(a[j], a[j+1]);
}
} }
*Sắp xếp chèn: INSERTIONSORT
void insertionsort(int a[], int n)
{
for(int i=1; i<n;i++)
{
int x=a[i], pos=i-1;
while(pos>=0 && x<a[pos])
{
a[pos+1]=a[pos];
--pos;
}
a[pos+1]=x;
} }
